func sprint05dfWire(field string, values []string, probes ...string) string {
	var wire bytes.Buffer
	wire.WriteString(http2.ClientPreface)
	fr := http2.NewFramer(&wire, nil)
	_ = fr.WriteSettings()
	var block bytes.Buffer
	enc := hpack.NewEncoder(&block)
	for i, value := range values {
		block.Reset()
		for _, f := range []hpack.HeaderField{{Name: ":method", Value: "GET"}, {Name: ":scheme", Value: "https"}, {Name: ":authority", Value: "example.com"}, {Name: ":path", Value: "/"}, {Name: field, Value: value}} {
			_ = enc.WriteField(f)
		}
		if len(probes) != 0 {
			_ = enc.WriteField(hpack.HeaderField{Name: "x-aegis-probe-id", Value: probes[i]})
		}
		_ = fr.WriteHeaders(http2.HeadersFrameParam{StreamID: uint32(2*i + 1), BlockFragment: block.Bytes(), EndHeaders: true, EndStream: true})
	}
	return hex.EncodeToString(wire.Bytes())
}

func sprint05dfResponse(statuses []string) string {
	var wire, block bytes.Buffer
	fr := http2.NewFramer(&wire, nil)
	_ = fr.WriteSettings()
	enc := hpack.NewEncoder(&block)
	for i, status := range statuses {
		block.Reset()
		_ = enc.WriteField(hpack.HeaderField{Name: ":status", Value: status})
		_ = fr.WriteHeaders(http2.HeadersFrameParam{StreamID: uint32(2*i + 1), BlockFragment: block.Bytes(), EndHeaders: true, EndStream: true})
	}
	return hex.EncodeToString(wire.Bytes())
}

func TestSprint05DFRuntimeBoundary(t *testing.T) {
	requireFixtureLock(t)
	key := bytes.Repeat([]byte{0x42}, 32)
	baseline := runtimeEvidence{
		Requests: []runtimeRequest{
			{Probe: "p0", Nonce: "n0", SNI: "example.com", Tenant: "fixture-A", Generation: 7, Status: 200, OriginHandlers: 1, GotConnection: true, Wrote: true},
			{Probe: "p1", SNI: "example.com", Tenant: "fixture-A", Generation: 7, Status: 421},
			{Probe: "p2", Nonce: "n2", SNI: "example.com", Tenant: "fixture-A", Generation: 7, Status: 200, OriginHandlers: 1, GotConnection: true, Wrote: true},
		},
		Wires: []runtimeWire{
			{ID: "downstream", SNI: "example.com", Verified: true, Closed: true, To: sprint05dfWire("x-aegis-probe-id", []string{"p0", "p1", "p2"}), From: sprint05dfResponse([]string{"200", "421", "200"})},
			{ID: "upstream", SNI: "example.com", Verified: true, Closed: true, To: sprint05dfWire("x-aegis-fixture-id", []string{"n0", "n2"}, "p0", "p2"), From: sprint05dfResponse([]string{"200", "200"})},
		},
	}
	for _, kind := range []string{"honest-bounded", "swapped-nonces", "duplicate-nonce-orphan-stream", "omitted-upstream-probe", "omitted-response-capture", "malformed-response-capture", "control-only-response-capture", "forged-rejection-response", "omitted-downstream-request", "omitted-origin-stream", "missing-TLS-verification", "wrong-generation", "capture-lost", "MAC-tamper"} {
		t.Run(kind, func(t *testing.T) {
			serialized, _ := json.Marshal(baseline)
			var e runtimeEvidence
			_ = json.Unmarshal(serialized, &e)
			switch kind {
			case "swapped-nonces":
				e.Requests[0].Nonce, e.Requests[2].Nonce = e.Requests[2].Nonce, e.Requests[0].Nonce
			case "duplicate-nonce-orphan-stream":
				e.Requests[2].Nonce = e.Requests[0].Nonce
			case "omitted-upstream-probe":
				e.Wires[1].To = sprint05dfWire("x-aegis-fixture-id", []string{"n0", "n2"})
			case "omitted-response-capture":
				e.Wires[0].From, e.Wires[1].From = "", ""
			case "malformed-response-capture":
				e.Wires[0].From, e.Wires[1].From = "not-hex", "00"
			case "forged-rejection-response":
				e.Wires[0].From = sprint05dfResponse([]string{"200", "200", "200"})
			case "control-only-response-capture":
				e.Wires[0].From, e.Wires[1].From = sprint05dfResponse(nil), sprint05dfResponse(nil)
			case "omitted-downstream-request":
				e.Requests = append(e.Requests[:1], e.Requests[2:]...)
			case "omitted-origin-stream":
				e.Wires[1].To = sprint05dfWire("x-aegis-fixture-id", []string{"n0"})
			case "missing-TLS-verification":
				e.Wires[1].Verified = false
			case "wrong-generation":
				e.Requests[0].Generation++
			case "capture-lost":
				e.Wires[0].Lost = true
			}
			e.MAC = runtimeMAC(e, key)
			if kind == "MAC-tamper" {
				e.Requests[0].Tenant = "forged"
			}
			got := assessRuntime(e, key)
			if got.RuntimeComplete || got.RuntimePass || got.Module01Pass || len(got.Incomplete) == 0 {
				t.Fatalf("incomplete fixture promoted to runtime PASS: %+v", got)
			}
			if kind == "honest-bounded" && !got.CorrelationConsistent {
				t.Fatalf("honest correlation positive failed: %+v", got)
			}
			b, _ := json.Marshal(map[string]any{"case": kind, "correlationConsistent": got.CorrelationConsistent, "runtimeComplete": got.RuntimeComplete, "runtimePass": got.RuntimePass, "incomplete": got.Incomplete})
			t.Log(string(b))
			if kind != "honest-bounded" && got.CorrelationConsistent {
				t.Fatalf("forged or omitted mapping accepted as consistent: %s", kind)
			}
		})
	}
}
